def find_relevant_duplications(self, query: str) -> list[DuplicatedPattern]:
        """
        Encuentra duplicaciones relevantes a la query del usuario.

        Esto ayuda a advertir: "Cuidado, ya hay código duplicado haciendo
        esto - no agregues otra copia, mejor refactoriza primero."
        """
        all_duplications = self.scan_project()

        query_lower = query.lower()
        keywords = set(re.findall(r'\w+', query_lower))

        relevant = []
        for dup in all_duplications:
            desc_lower = dup.description.lower()
            desc_keywords = set(re.findall(r'\w+', desc_lower))

            if keywords & desc_keywords:  # Intersección no vacía
                relevant.append(dup)

        return relevant

def get_duplication_warnings(self, query: str) -> list[str]:
        """Genera advertencias sobre duplicación para incluir en el contexto."""
        relevant = self.find_relevant_duplications(query)

        warnings = []
