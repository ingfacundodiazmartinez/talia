import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:talia/utils/release_logger.dart';

/// Servicio para optimizar queries a Firestore evitando N+1
class FirestoreBatchService {
  static final FirestoreBatchService _instance = FirestoreBatchService._internal();
  factory FirestoreBatchService() => _instance;
  FirestoreBatchService._internal();

  final FirebaseFirestore _firestore = FirebaseFirestore.instance;

  /// Obtener múltiples documentos por IDs en una sola query
  Future<Map<String, DocumentSnapshot>> getDocumentsByIds(
    String collection,
    List<String> ids, {
    int batchSize = 30, // Firestore limita a 30 documentos por query "in"
  }) async {
    try {
      final Map<String, DocumentSnapshot> result = {};
      
      // Procesar en batches debido a la limitación de Firestore (30 documentos por query "in")
      for (int i = 0; i < ids.length; i += batchSize) {
        final batchIds = ids.sublist(i, i + batchSize > ids.length ? ids.length : i + batchSize);
        
        if (batchIds.isEmpty) continue;
        
        final querySnapshot = await _firestore
            .collection(collection)
            .where(FieldPath.documentId, whereIn: batchIds)
            .get();
        
        for (final doc in querySnapshot.docs) {
          result[doc.id] = doc;
        }
        
        // Pequeño delay entre batches para evitar rate limiting
        if (i + batchSize < ids.length) {
          await Future.delayed(const Duration(milliseconds: 100));
        }
      }
      
      ReleaseLogger.info('Batch query completada: ${result.length}/${ids.length} documentos obtenidos', 
          tag: 'FirestoreBatchService');
      return result;
    } catch (e) {
      ReleaseLogger.error('Error en batch query: $e', tag: 'FirestoreBatchService');
      return {};
    }
  }

  /// Verificar existencia de múltiples documentos
  Future<Map<String, bool>> checkDocumentsExist(
    String collection,
    List<String> ids, {
    String field = 'id',
  }) async {
    try {
      final Map<String, bool> result = {};
      
      // Agrupar por valores únicos para usar whereIn
      final uniqueValues = ids.toSet().toList();
      
      for (int i = 0; i < uniqueValues.length; i += 30) {
        final batchValues = uniqueValues.sublist(
          i, 
          i + 30 > uniqueValues.length ? uniqueValues.length : i + 30
        );
        
        final querySnapshot = await _firestore
            .collection(collection)
            .where(field, whereIn: batchValues)
            .get();
        
        // Crear set de valores existentes
        final existingValues = querySnapshot.docs
            .map((doc) => doc.data()[field]?.toString())
            .where((value) => value != null)
            .toSet();
        
        for (final value in batchValues) {
          result[value] = existingValues.contains(value);
        }
      }
      
      return result;
    } catch (e) {
      ReleaseLogger.error('Error verificando existencia: $e', tag: 'FirestoreBatchService');
      return {};
    }
  }

  /// Obtener documentos por campo específico en batch
  Future<List<QueryDocumentSnapshot>> getDocumentsByFieldValues(
    String collection,
    String field,
    List<dynamic> values, {
    int limit = 100,
  }) async {
    try {
      final List<QueryDocumentSnapshot> results = [];
      
      for (int i = 0; i < values.length; i += 30) {
        final batchValues = values.sublist(
          i, 
          i + 30 > values.length ? values.length : i + 30
        );
        
        final querySnapshot = await _firestore
            .collection(collection)
            .where(field, whereIn: batchValues)
            .limit(limit)
            .get();
        
        results.addAll(querySnapshot.docs);
      }
      
      return results;
    } catch (e) {
      ReleaseLogger.error('Error obteniendo documentos por campo: $e', tag: 'FirestoreBatchService');
      return [];
    }
  }
}

async function cleanChatMessages() {
  console.log('\n🔍 Limpiando mensajes de chats demo...');

  const chatsSnapshot = await db.collection('chats').get();
  let deletedChats = 0;

  // Obtener todos los IDs de participantes únicos
  const allParticipantIds = new Set();
  const chatParticipantsMap = new Map(); // chatId -> [participantIds]

  chatsSnapshot.docs.forEach(chatDoc => {
    const participants = chatDoc.data().participants || [];
    chatParticipantsMap.set(chatDoc.id, participants);
    participants.forEach(id => allParticipantIds.add(id));
  });

  // BATCH QUERY: Obtener todos los usuarios demo en una sola consulta
  const participantIdsArray = Array.from(allParticipantIds);
  const userDocsMap = new Map();

  // Procesar en batches de 30 (límite de Firestore para whereIn)
  for (let i = 0; i < participantIdsArray.length; i += 30) {
    const batchIds = participantIdsArray.slice(i, i + 30);
    
    const usersSnapshot = await db.collection('users')
      .where(FieldPath.documentId, 'in', batchIds)
      .get();

    usersSnapshot.docs.forEach(userDoc => {
      userDocsMap.set(userDoc.id, userDoc);
    });
  }

  // Identificar chats demo
  const demoChatIds = [];
  for (const [chatId, participants] of chatParticipantsMap) {
    let isDemoChat = false;
    
    // Verificar sin queries adicionales
    for (const participantId of participants) {
      const userDoc = userDocsMap.get(participantId);
      if (userDoc && userDoc.data().isDemo === true) {
        isDemoChat = true;
        break;
      }
    }

    if (isDemoChat) {
      demoChatIds.push(chatId);
    }
  }

  // Eliminar mensajes y chats en batch
  console.log(`🗑️  Eliminando ${demoChatIds.length} chats demo...`);
  
  for (const chatId of demoChatIds) {
    const chatRef = db.collection('chats').doc(chatId);
    
    // Eliminar mensajes en subcolección
    const messagesSnapshot = await chatRef.collection('messages').get();
    const messageBatch = db.batch();
    messagesSnapshot.docs.forEach(msg => messageBatch.delete(msg.ref));
    await messageBatch.commit();

    // Eliminar chat
    await chatRef.delete();
    deletedChats++;
  }

  console.log(`✓ Eliminados ${deletedChats} chats demo`);
}

  }
}
