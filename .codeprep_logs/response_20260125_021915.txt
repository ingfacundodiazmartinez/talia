import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import '../utils/release_logger.dart';

/// Servicio para obtener datos de múltiples usuarios en una sola query
class UserBatchService {
  final FirebaseFirestore _firestore;
  final FirebaseAuth _auth;
  
  static final Map<String, Map<String, dynamic>> _userCache = {};
  static DateTime? _lastCacheCleanup;
  
  UserBatchService({
    FirebaseFirestore? firestore,
    FirebaseAuth? auth,
  }) : _firestore = firestore ?? FirebaseFirestore.instance,
       _auth = auth ?? FirebaseAuth.instance;
  
  /// Obtener múltiples usuarios en una sola query (máx 10 por batch)
  Future<Map<String, Map<String, dynamic>>> getUsersBatch(List<String> userIds) async {
    if (userIds.isEmpty) return {};
    
    // Filtrar usuarios ya en cache
    final uncachedIds = <String>[];
    final result = <String, Map<String, dynamic>>{};
    
    for (final userId in userIds) {
      if (_userCache.containsKey(userId)) {
        result[userId] = _userCache[userId]!;
      } else {
        uncachedIds.add(userId);
      }
    }
    
    // Si todos están en cache, retornar
    if (uncachedIds.isEmpty) return result;
    
    // Firestore limita a 10 documentos en "in" queries
    const batchSize = 10;
    for (var i = 0; i < uncachedIds.length; i += batchSize) {
      final batchIds = uncachedIds.sublist(
        i,
        i + batchSize > uncachedIds.length ? uncachedIds.length : i + batchSize,
      );
      
      try {
        final snapshot = await _firestore
            .collection('users')
            .where(FieldPath.documentId, whereIn: batchIds)
            .get();
        
        for (final doc in snapshot.docs) {
          final userData = doc.data();
          _userCache[doc.id] = userData;
          result[doc.id] = userData;
        }
      } catch (e) {
        ReleaseLogger.error('Error en batch query de usuarios: $e', tag: 'UserBatchService');
        // Fallback: obtener uno por uno (solo si es crítico)
        for (final userId in batchIds) {
          try {
            final doc = await _firestore.collection('users').doc(userId).get();
            if (doc.exists) {
              final userData = doc.data() as Map<String, dynamic>;
              _userCache[userId] = userData;
              result[userId] = userData;
            }
          } catch (fallbackError) {
            ReleaseLogger.error('Error obteniendo usuario $userId: $fallbackError', tag: 'UserBatchService');
          }
        }
      }
    }
    
    // Limpiar cache periódicamente
    _cleanupCache();
    
    return result;
  }
  
  /// Obtener un solo usuario (usa cache si está disponible)
  Future<Map<String, dynamic>?> getUser(String userId) async {
    if (_userCache.containsKey(userId)) {
      return _userCache[userId];
    }
    
    try {
      final doc = await _firestore.collection('users').doc(userId).get();
      if (doc.exists) {
        final userData = doc.data() as Map<String, dynamic>;
        _userCache[userId] = userData;
        _cleanupCache();
        return userData;
      }
    } catch (e) {
      ReleaseLogger.error('Error obteniendo usuario $userId: $e', tag: 'UserBatchService');
    }
    
    return null;
  }
  
  /// Invalidar cache de un usuario específico
  void invalidateUserCache(String userId) {
    _userCache.remove(userId);
  }
  
  /// Limpiar cache (mantiene solo los últimos 100 usuarios)
  void _cleanupCache() {
    final now = DateTime.now();
    if (_lastCacheCleanup != null && 
        now.difference(_lastCacheCleanup!).inMinutes < 5) {
      return;
    }
    
    if (_userCache.length > 100) {
      final keysToRemove = _userCache.keys.toList().sublist(0, _userCache.length - 100);
      for (final key in keysToRemove) {
        _userCache.remove(key);
      }
    }
    
    _lastCacheCleanup = now;
  }
}

// Añadir al final de la clase ListMessagesService:

  /// Buscar mensajes por texto usando índices (más eficiente)
  /// Requiere crear índice compuesto en Firestore: messages_search_index
  /// Campos: chatId (ascending), timestamp (descending), text (ascending)
  Future<List<ChatMessage>> searchByTextEfficient({
    required String chatId,
    required String query,
    bool isGroup = false,
    int limit = 50,
  }) async {
    final userId = _auth.currentUser?.uid;
    if (userId == null) return [];
    
    if (query.trim().isEmpty) return [];
    
    try {
      final collection = isGroup ? 'groups_v2' : 'chats';
      
      // Usar índice para búsqueda más eficiente
      // Nota: Firestore no soporta LIKE, pero podemos usar >= y <= para prefijos
      final normalizedQuery = query.toLowerCase();
      final endQuery = normalizedQuery + '\uf8ff'; // Carácter Unicode alto
      
      final snapshot = await _firestore
          .collection(collection)
          .doc(chatId)
          .collection('messages')
          .where('searchableText', isGreaterThanOrEqualTo: normalizedQuery)
          .where('searchableText', isLessThanOrEqualTo: endQuery)
          .orderBy('searchableText')
          .orderBy('timestamp', descending: true)
          .limit(limit)
          .get();
      
      return snapshot.docs
          .map((doc) => ChatMessage.fromFirestore(doc, currentUserId: userId))
          .toList();
    } catch (e) {
      ReleaseLogger.error('Error en búsqueda eficiente: $e', tag: 'ListMessages');
      // Fallback a búsqueda tradicional
      return searchByText(
        chatId: chatId,
        query: query,
        isGroup: isGroup,
        limit: limit,
      );
    }
  }

  /// Obtener múltiples mensajes por IDs en batch
  Future<List<ChatMessage>> getMessagesBatch({
    required String chatId,
    required List<String> messageIds,
    bool isGroup = false,
  }) async {
    final userId = _auth.currentUser?.uid;
    if (userId == null) return [];
    
    if (messageIds.isEmpty) return [];
    
    try {
      final collection = isGroup ? 'groups_v2' : 'chats';
      
      // Firestore limita a 10 documentos en "in" queries
      const batchSize = 10;
      final allMessages = <ChatMessage>[];
      
      for (var i = 0; i < messageIds.length; i += batchSize) {
        final batchIds = messageIds.sublist(
          i,
          i + batchSize > messageIds.length ? messageIds.length : i + batchSize,
        );
        
        final snapshot = await _firestore
            .collection(collection)
            .doc(chatId)
            .collection('messages')
            .where(FieldPath.documentId, whereIn: batchIds)
            .get();
        
        final batchMessages = snapshot.docs
            .map((doc) => ChatMessage.fromFirestore(doc, currentUserId: userId))
            .toList();
        
        allMessages.addAll(batchMessages);
      }
      
      return allMessages;
    } catch (e) {
      ReleaseLogger.error('Error obteniendo batch de mensajes: $e', tag: 'ListMessages');
      return [];
    }
  }
