class ChatRepository {
  final FirebaseFirestore _firestore;
  final FirebaseAuth _auth;

  ChatRepository({
    required FirebaseFirestore firestore,
    required FirebaseAuth auth,
  }) : _firestore = firestore,
       _auth = auth;

  String? get currentUserId => _auth.currentUser?.uid;

  /// Obtener lista de chats del usuario
  Stream<QuerySnapshot> watchUserChats({
    int limit = 50,
    bool includeGroups = true,
  }) {
    final userId = currentUserId;
    if (userId == null) {
      return const Stream.empty();
    }

    return _firestore
        .collection('users')
        .doc(userId)
        .collection('chats')
        .orderBy('lastMessageAt', descending: true)
        .limit(limit)
        .snapshots();
  }

  /// Crear chat 1:1 (solo Cloud Functions debería hacer esto)
  Future<String> createDirectChat({
    required String otherUserId,
    Map<String, dynamic>? initialMetadata,
  }) async {
    try {
      final currentUser = _auth.currentUser;
      if (currentUser == null) throw Exception('Usuario no autenticado');

      // Solo Cloud Functions puede crear chats
      // Este método sería para casos especiales o testing
      final chatData = {
        'participants': [currentUser.uid, otherUserId],
        'createdAt': FieldValue.serverTimestamp(),
        'updatedAt': FieldValue.serverTimestamp(),
        'type': 'direct',
        'lastMessage': '',
        'lastMessageAt': FieldValue.serverTimestamp(),
        'createdBy': currentUser.uid,
      };

      if (initialMetadata != null) {
        chatData.addAll(initialMetadata);
      }

      final chatRef = await _firestore.collection('chats').add(chatData);
      
      // Agregar referencia a ambos usuarios
      final batch = _firestore.batch();
      
      batch.set(
        _firestore.collection('users').doc(currentUser.uid).collection('chats').doc(chatRef.id),
        {'chatId': chatRef.id, 'updatedAt': FieldValue.serverTimestamp()}
      );
      
      batch.set(
        _firestore.collection('users').doc(otherUserId).collection('chats').doc(chatRef.id),
        {'chatId': chatRef.id, 'updatedAt': FieldValue.serverTimestamp()}
      );
      
      await batch.commit();
      
      return chatRef.id;
    } catch (e) {
      ReleaseLogger.error('Error creando chat directo: $e', tag: 'ChatRepository');
      throw Exception('Error creando chat: $e');
    }
  }

  /// Obtener metadata específica de un chat

  /// Verificar si existe un chat entre dos usuarios
  Future<String?> findExistingChat(String otherUserId) async {
    try {
      final currentUser = _auth.currentUser;
      if (currentUser == null) return null;

      // Buscar en chats donde ambos sean participantes
      final query = await _firestore
          .collection('chats')
          .where('participants', arrayContains: currentUser.uid)
          .where('type', isEqualTo: 'direct')
          .get();

      for (final doc in query.docs) {
        final participants = List<String>.from(doc.data()['participants'] ?? []);
        if (participants.contains(otherUserId)) {
          return doc.id;
        }
      }
      
      return null;
    } catch (e) {
      ReleaseLogger.error('Error buscando chat existente: $e', tag: 'ChatRepository');
      return null;
    }
  }

  Future<List<String>> getChatParticipants({
    required String chatId,
    bool isGroup = false,
  }) async {
    try {
      final collection = isGroup ? 'groups_v2' : 'chats';
      final doc = await _firestore.collection(collection).doc(chatId).get();
      
      if (!doc.exists) return [];
      
      final data = doc.data();
      if (data == null) return [];
      
      return List<String>.from(data['participants'] ?? []);
    } catch (e) {
      ReleaseLogger.error('Error obteniendo participantes: $e', tag: 'ChatRepository');
      return [];
    }
  }

          });
    } catch (e) {
      ReleaseLogger.error('Error actualizando metadata: $e', tag: 'ChatRepository');
      throw Exception('Error actualizando metadata: $e');
    }

### 3. **Consideraciones de Arquitectura**

1. **Separación de responsabilidades**:
   - `ChatRepository`: Operaciones de chat (metadata, participantes, preferencias)
   - `MessageRepository`: Operaciones de mensajes (CRUD de mensajes)
   - `ChatMessagingService`: Lógica de negocio para enviar mensajes

2. **Evitar duplicación**:
   - No recrear métodos de mensajes que ya existen
   - Usar `MessageRepository` para todo lo relacionado con mensajes
   - `ChatRepository` solo para metadata del chat

3. **Consistencia con patrones existentes**:
   - Usar `ReleaseLogger` para logging
   - Implementar `FieldValue.serverTimestamp()` para timestamps
   - Manejar errores de forma consistente
